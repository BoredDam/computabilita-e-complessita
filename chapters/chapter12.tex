	\chapter{Programmi universali}
	
	\section{Programma universale}
	Si consideri la funzione $\psi(x,y) =_{def} \phi_x(y)$. Questa funzione è universale, in quanto, ponendo $g_m(y) =_{def} \psi(m,y)$, la sequenza di $g_0, g_1, g_2, \dots$ contiene tutte le funzioni unarie calcolabili $\phi_0^{(1)}, \phi_1^{(1)}, \phi_2^{(1)}, \dots$.
	In notazione lambda:
	\begin{equation}
	\begin{aligned}
	g_m &= \lambda y.\psi(m,y)\\
	&= \lambda y.\phi_m(y)\\
	&= \phi_m
	\end{aligned}
	\end{equation}
	
	\subsection{Calcolabilità della funzione universale}
	La funzione universale per le funzioni $n$-arie calcolabili è una funzione $(n+1)$-aria $\psi_U^{(n)}$ calcolabile $\forall n \geq 1$, ed è definita da:
	
	\begin{equation}
	\psi_U^{(n)}(e, x_1, \dots, x_n) = \phi_e^{(n)}(x_1, \dots, x_n)
	\end{equation}
	
	\paragraph{Dimostrazione informale della calcolabilità}
	Fissiamo $n \geq 1$. Dati un indice $e$ e una $n$-upla $\vec{x} = x_1, \dots, x_n$, possiamo calcolare $\phi_U^{(n)}(e, \vec{x})$ nel seguente modo:
	\begin{enumerate}
		\item Si costruisca il programma $P_e$
		\item Si simuli $P_e$ sull'input $\vec{x}$
	\end{enumerate}
	%
	Essendo	la procedura precedente una procedura univoca e ben definita,  $\phi_U^{(n)}$ è calcolabile per la tesi di Church.

	\section{Predicati di verifica della fermata limitata}
	I seguenti predicati sono decidibili:
	\begin{itemize}
		\item $S_n(e, \vec{x}, y, t) \cong \text{"$P_e(\vec{x})\downarrow y$ in al più $t$ passi."}$
		\item $T_n(e, \vec{x}, y, t) \cong \text{"$P_e(\vec{x})\downarrow$ in al più $t$ passi."}$
	\end{itemize}
	
	\subsection{Teorema della forma normale di Kleene}
	Esistono una funzione calcolabile $U(x)$, e , per ogni $n \geq 1$, un predicato decidibile $T_n(e, \vec{x}, z)$ tali che:
	\begin{enumerate}
		\item $\phi_e^{(n)}(\vec{x})\downarrow$ se e solo se $\exists z T_n(e, \vec{x}, z) = 1$
		\item $\phi_e^{(n)}(\vec{x}) = U(\mu zT_n(e, \vec{x}, z))$
	\end{enumerate}
	%
	Un risultato molto interessante di questo teorema, è legato alle funzioni $S_n(e, \vec{x}, y, t), \, T_n(e, \vec{x}, z)$, che sono $\mathcal{PR}$. Il teorema della forma normale dice quindi che ogni funzione calcolabile può esser eottenuta mediante \textbf{al più una sola} applciazione dell'operatore di minimalizzazione $\mu$.
	
	
	\subsection{Definizione per casi \textit{con funzioni parziali!}}
	Questa possibilità di simulare le funzioni fino al $t$-esimo passo, permette di gestire anche definizioni per casi che includono funzioni parziali. Invece di usare la sommatoria di $f(\vec{x}) \cdot c_M(\vec{x})$, si calcolano in parallelo tutte le funzioni, un passo alla volta. Quando la funzione a cui è associato il predicato soddisfatto termina, allora termina l'intera funzione definita per casi.
	
	\section{Indecidibilità di "\textit{$\phi_x$ è totale}"}
	Il problema ``$\phi_x$ è totale'' è indecidibile. 
	\paragraph{Dimostrazione}
	Sia $M(x) := \text{``$\phi_x$ è totale''}$. Se $M(x)$ fosse decidibile, allora ponen
	
	\begin{equation}
	f(x) := \begin{cases}
	\psi_U(x,x) + 1 & \text{se ``$\phi_x$ è totale''}\\
	0 & \text{se ``$\phi_x$ non è totale''}\\
	\end{cases}
	\end{equation}
	%
	la funzione $f$ sarebbe calcolabile, pertanto esiste $e \in \mathbb{N}$ tale che $f= \phi_e$. $f$ è totale, quindi $\phi_e$ è totale, dunque:
	$$
	\phi_e(e) = f(e) = \psi_U(e,e) + 1 = \phi_e(e) +1
	$$
	\section{Operazioni effettiva su funzioni calcolabili}
	Ci chiediamo se sia possibile manipolare in maniera effettiva funzioni o insiemi infiniti (ricorsivamente enumerabili\footnote{Gli insiemi ricorsivamente enumerabili sono collezioni di elementi (solitamente numeri naturali) per i quali esiste un algoritmo capace di elencarne i membri o di riconoscere quelli validi in tempo finito}): in alcuni casi, questo è possibile, tramite applicazione del teorema di parametrizzazione!
	
	\subsection{Esempio 1}
	Date $\phi_x, \phi_y$, calcolare un indice di $\phi_x\phi_y$.
	
	$$
	\begin{aligned}
		\phi_x(z)\cdot \phi_y(z)
		&=
		\psi_U(y_1,z)\cdot \psi_U(y,z) \\
		&= \phi^{(3)}_e(x,y,z) \\
		&= \phi_{S_1^2(e,x,y)}(z) \\
		&= \phi_{k(x,y)}(z)
		\quad
		\big(\text{con } k(x,y):=S_1^2(e,x,y)\ \text{totale calcolabile}\big) \\
		&\Longrightarrow\quad
		\phi_x\cdot \phi_y = \phi_{k(x,y)}
	\end{aligned}
	$$
	
		\subsection{Esempio 2}
		Esiste una funzione calcolabile totale $g(x)$ tale che $(\phi_x)^2 = \phi_{g(x)}$.
	
	$$
	\begin{aligned}
		\phi_x(z)\cdot \phi_y(z)
		&=
		\psi_U(y_1,z)\cdot \psi_U(y,z) \\
		&= \phi^{(3)}_e(x,y,z) \\
		&= \phi_{S_1^2(e,x,y)}(z) \\
		&= \phi_{k(x,y)}(z)
		\quad
		\big(\text{con } k(x,y):=S_1^2(e,x,y)\ \text{totale calcolabile}\big) \\
		&\Longrightarrow\quad
		\phi_x\cdot \phi_y = \phi_{k(x,y)}
	\end{aligned}
	$$
